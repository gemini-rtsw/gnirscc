/*
 *+
 * FUNCTION NAME:
 * 	saveDataset
 *
 * INVOCATION:
 * 	spawned at startup
 *
 * PARAMETERS: (">" input, "!" modified, "<" output)
 *	none
 *
 * FUNCTION VALUE:
 * 	 Only returns on error.
 *
 * PURPOSE:
 * 	Wait for datacoadder board to signal that a frame is ready to be saved and then
 *		save it.
 *
 * DESCRIPTION:
 * 
 *
 * EXTERNAL VARIABLES:
 * 	svrP - saver parameters
 *	sem - semaphores
 *	obsDone - done signal
 *
 * PRIOR REQUIREMENTS:
 * None
 *
 * DEFICIENCIES:
 * none
 *
 * HISTORY:
 *  written by Peter Ruckle
 *-
 */
int saveDataset()
{
#ifdef DEBUG
    int t1 = 1;
#endif
    int i;
    int buf;
    long status;
    roiParams *proi;
    char errString[80];
    DHS_STATUS rc = DHS_S_SUCCESS;
	dhsParams *dhs;
	
    sleep(5,0);
    /* connect to server if we are using dhs*/
    if(svrP.server == DHS_SAVE)    {
	dhs = &(svrP.param.dhs);
	
	printf("connecting to dhs name = %s\n",dhs->name);

	saverOK = 0;
	if(DhsConnect(&svrP.param.dhs) != OK)	{
		rc = 0;

    	if(gnPutEpicsT(dbTop, DHSCONNECTED ".VAL", DCALONG, &rc)!= OK)
    			printf ("failed to set DHSCONNECTED\n");

	    printf("couldn't connect to dhs\n");
	    printf("Cpu needs to be rebooted\n");
	    return ERROR;
	}
	
/* 
 *     dhsCallbackSet(DHS_CBT_PUT, dhsPutCallback, &rc);
 *     printf( "----->After setting put callback, dhsStatus is %d \n", rc);
 */
 
 

	printf("connected\n");


	rc = 1;

  if(gnPutEpicsT(dbTop, DHSCONNECTED ".VAL", DCALONG, &rc)!= OK)
    		printf ("failed to set DHSCONNECTED\n");
 	
	
	firstTick = tickGet();
	
    }
    while (1)    {
	/* wait for semaphore, this is released by the unscramble interrupt
	   routine  (a frame is ready to be saved)*/
/* 	printf("Waiting for frame to save\n"); */
	if(semTake (sem.unScrambleFrame,WAIT_FOREVER) == ERROR)
	{
	    /* print fatal error*/
	    printf("saveDataset: Error taking semaphore\n");
	    goto Error;
	}
	
	
	setObsFlags(FALSE, FALSE, TRUE); /*set prep FALSE, acq FALSE, rdout TRUE */

		if (dhs->headers==PARTIALHEAD) {
			gemLogMsg(2,"not collecting headers\n");
			}
		else {
			gemLogMsg(2,"collecting headers\n");
			}

	
	
	gemLogMsg( 5 ,"saver task got semaphore\n");

	/* get buffer to put data in */
	buf = getFullBuf();

	/* check to see if observation was aborted*/
	if(DCA_Save  == FALSE)
	{
	    printf("observation aborted\n");
	    if(buf != ERROR)
		putEmptyBuf(buf);
		setObsFlags(FALSE, FALSE, FALSE);
		continue;
	}
	/* this should never happen unless the coadder hardware is broke, or
	 if sdt mode did not kill the interrupts */
	if(buf == ERROR)
	{
	    printf("error getting full buffer\n");
		continue;
	}
		
	if (dhs->headers != PARTIALHEAD)
		status = getDsHeader(AFTER,buf);
	
	/*  roi info*/
	proi = &svrP.roi;

	/* dhs or fits*/
	if(svrP.server == DHS_SAVE)
	{
	gemLogMsg( 5 ,"saveDhsData for %d\n",proi->numRois);
	status = saveDhsData(buf,&(svrP.param.dhs), proi) ;
	}
	else  /* wrong image type*/ {
	    printf("Only allow dhs images %d\n",svrP.server);
		}
	
	if(status != OK)
	{
	    /* print error*/
	    sprintf(errString,"saveDataset: Data not saved \n");
	    printf(errString);
	    /* set car record*/ 
	    sleep(1,0);
	     setCar(OBSERVE_CAR,menuCarstatesERROR,ERROR,errString,dummy);
	   
	   
	} 

	setObsFlags(FALSE, FALSE, FALSE); /*set prep FALSE, acq FALSE, rdout FALSE */


    }
  Error:
  	saverOK = 0;
 if(gnPutEpicsT(dbTop, DHSCONNECTED ".VAL", DCALONG, &saverOK)!= OK)
  		 printf ("failed to set DHSCONNECTED\n");

    /*print error*/
    printf("saveDataset returned error:  exiting\n");
    printf("Cpu needs to be rebooted\n");
    return ERROR;
}

    /* add connect to headers and initialization ????? */
    status = DHS_S_SUCCESS;


    /* connect to dhs server if not done already*/
    if (dhs->connect == NULL)
    {
	gemLogMsg( 5,"connecting to dhs name = %s from %s\n",dhs->name,dhs->impName);
	 
	if(DhsConnect(dhs) != OK)
	{
	    printf("couldn't connect to dhs\n");
	    goto DHSError1;
	} 
	if (*(int *)dhs->connect == NULL)
	{
	    sprintf(errString,"saveDhsData: ERROR: saveFrame - unable to initialize Saver connection\n");
	   
	    printf(errString);
	    goto DHSError1;
	}
	gemLogMsg( 5 ,"connected to dhs\n");
    }

    status = DHS_S_SUCCESS;

    /* copy structure to local area*/
    memcpy(&cb.roi,roiStruct,sizeof(roiParams));
    memcpy(&cb.dhs,dhs,sizeof(dhsParams));
    cb.bufNum = bufNum;
    cb.run = run;


/*
 *     sprintf(name,"tdhsSave%d",run);
 *     status =   taskSpawn(name,60,VX_FP_TASK,100000,dhsSave,&cb,
 *         	 0,0,0,0,0,0,0,0,0);
 *     if(status == ERROR) {
 *       printf("couldn't spawn task \n");
 *           goto DHSError1;
 *           }
 */
    dhsSave((int) &cb,0,0,0,0,0,0,0,0,0);
  	return OK;

  DHSError1:
    setCar(OBSERVE_CAR,menuCarstatesERROR,ERROR,errString,dummy);
   
    printf("Error in saveDhsData\n");
    status = DHS_S_SUCCESS;
    /*     dhsDisconnect(dhs->connect,&status); */
    status = DHS_S_SUCCESS;
    dhsExit(&status);
    if(status != DHS_S_SUCCESS)
	printf("dhsExit failed\n");
    dhs->initDone = 0;
    dhs->connect = NULL;

    return ERROR;
}

/* global variable to signal error in another save process */
static int dhs_ok=1;

int dhsSave(int arg1 ,int arg2 ,int arg3 ,int arg4 ,int arg5 ,int arg6 ,
	    int arg7 ,int arg8 ,int arg9 ,int arg10 )
{ 
    int size;      /* size of current data to be sent*/
    DHS_TAG putTag;
    DHS_BD_DATASET ds = NULL;
#ifdef DEBUG
    int t1 = 1;/* debugging flag*/
#endif
    DHS_STATUS status = DHS_S_SUCCESS;
    DHS_BD_FRAME frame;
    int i = 0;
	
	unsigned long ndims[1];
    int dx,dy,index;
    unsigned long dims[2];
    unsigned long origin[2];
	
    char *qlStreams[1];
	
	int missing=0;
	
    char *contrib[1];
    char *vme;           /* vme address of local buffer*/
    char errString[80];
    char label[MAXSTRING]; 
    char *dsName;   		/* dataset name*/
    char dsDHSdataLabel[MAXSTRING];   /* holder for OCS provided dataset name */
    int run;   /* current frame number, for debugging only*/
    int bufNum;
    callBackStruct *cb = NULL;/* structure passed from caller*/
    dhsParams *dhs;   /* dhs structure*/
    roiParams *roiStruct; /* roi structure*/
    tDCARegs reg;/* coadder register structure*/
    tRect *proi;

#ifdef TRACE
	dhs_timer *ptimer;
#endif

	int dhs_i;
	int dhs_j;
	long  *dhs_data32;
	
	long dhs_dataMAX, dhs_dataMIN;

    /* initialize variables*/
	
#ifdef TRACE
	ptimer = (dhs_timer*) malloc (sizeof(dhs_timer));
	ptimer->start = tickGet();
#endif
	
	origin[0] = 1;
    origin[1] = 1;

    contrib[0] = "2DIRS";
    cb = (callBackStruct *)arg1;
    run  = cb->run;
    dhs = &cb->dhs;
    roiStruct = &cb->roi;
    bufNum = cb->bufNum;
   
#ifdef TRACE
	gemTimeLogMsg(ptimer, 5 ,"%d, dhsSave task\n",run);
#else
	gemLogMsg(5 ,"%d, dhsSave task\n",run);
#endif


   /* check numPics for movie mode */
   gemLogMsg(1 ,"numPics %d\n",numPics);

    /* DHSdataLabel */
	/* if one is supplied - do not set the lifetime (assumed to be done by OCS) */
	/* if not, query one from DHS and set lifetime */
    if(dhs_ok) {
		  /* one supplied */
  	      if (strlen(dhs->dhsDatalabel) >= 1) {
		  		strncpy(dsDHSdataLabel,dhs->dhsDatalabel,MAX_STRING_SIZE);
		  		dsName = &dsDHSdataLabel[0];
				}
		  else {
			/* get it from the DHS server */
			dsName = dhsBdName(dhs->connect,&status);
			if(status != DHS_S_SUCCESS) {
				sprintf(errString,"saveDhsData: ERROR: saveFrame - unable to get image name\n");
				printf(errString);
				goto DHSError;
				}
			/* need set lifetime */
			if (dhs->lifetime == DHS_TEMP) {
				/* movie mode ... QlTool only - do not set any lifetime
				 * dhsBdCtl(dhs->connect, DHS_BD_CTL_LIFETIME,dsName,DHS_BD_LT_TRANSIENT, &status);
				 */
				}
			else
				dhsBdCtl(dhs->connect, DHS_BD_CTL_LIFETIME,dsName,DHS_BD_LT_PERMANENT, &status);
				if(status != DHS_S_SUCCESS) {
					sprintf(errString,"saveDhsData: ERROR: saveFrame - unable to set lifetime\n");
					printf(errString);
					goto DHSError;
					}
				}
		}

	



	
	if(strlen(dhs->qlStream) >0) {
		qlStreams[0]=dhs->qlStream;
		dhsBdCtl(dhs->connect, DHS_BD_CTL_QLSTREAM,dsName, 1,qlStreams , &status);
	    	if(status != DHS_S_SUCCESS)
	    	{
			sprintf(errString,"saveDhsData: ERROR: saveFrame - unable to set Ql stream\n");
			printf(errString);
			goto DHSError;
	    	}
		}
		
		
    gemLogMsg( 4 ,"%d handling %d rois\n",run,roiStruct->numRois);

	while(i < roiStruct->numRois)    {
    	/* create dataset*/
    	if(dhs_ok)
		ds = dhsBdDsNew(&status);
    	if (status != DHS_S_SUCCESS)
    	{
		sprintf(errString,"saveDhsData: ERROR: saveFrame - unable to create data set\n");

		printf(errString);
			goto DHSError;
    	}
    	gemLogMsg( 4 ,"%d created dataset %x \n",run,ds);

	if (dhs->headers != PARTIALHEAD) {
		getDsHeader(NOW,bufNum);
		/* write headers to dataset */
		getFrameHeader(NOW,bufNum);
		}




	
    gemLogMsg( 4 ,"%d handling roi #%d\n",run,i);

	proi = &roiStruct->roi[i];
	
	
	
	/*
	 * bufNum is not used but getWCSInfo !!!
	 */
	if (dhs->headers != PARTIALHEAD) {
		if( getWCSInfo(i,bufNum) == ERROR) {
	    	printf("error getting wcs info\n");
			}
		}


	/* this somehow was missing ... */
	if (dhs->headers != PARTIALHEAD) {
		setHeaderROI(i,proi,bufNum);
		}

	/* create frame - equ. of save fits should start here*/
	
	dx = proi->hiX - proi->lowX +1;
	dy = proi->hiY - proi->lowY +1;	

    /* calculate start address    */
    reg.pXferStart =  MAKE_TRANS_ADDR(bufNum , proi->lowX , proi->lowY) ;
    reg.numRows = dy ;
    reg.numCols = dx ;


	ndims[0] = 2;
	dims[0] = dx;
	dims[1] = dy;

	index = i;

	/* create label from ObservID*/
	sprintf(label,"%s:%d",dsName,index);
	
	if(dhs_ok)
		if (dhs->format == DHS_UINT32)
	    	frame = dhsBdFrameNew( ds, label, index, DHS_DT_INT32, ndims[0], dims, (const void **)&dhs->data,&status); 
		else if (dhs->format == DHS_UINT16) {
	    	frame = dhsBdFrameNew( ds, label, index, DHS_DT_INT16, ndims[0], dims, (const void **)&dhs->data16,&status);
			dhs->data = (long*) malloc (sizeof(long)*dx*dy);
			} 
		else if (dhs->format == DHS_UINT8) {
	    	frame = dhsBdFrameNew( ds, label, index, DHS_DT_INT8, ndims[0], dims, (const void **)&dhs->data8,&status);
			dhs->data = (long*) malloc (sizeof(long)*dx*dy);
			} 

    	if (status != DHS_S_SUCCESS)
    	{
		sprintf(errString,"saveDhsData: ERROR: unable to allocate new frame buffer\n");
		printf(errString);
		goto DHSError;
    	}

	/* this are compulsary keyword */		
	dhsBdAttribAdd (ds, "instrument", DHS_DT_STRING, 0, NULL, "NIRI" ,&status );
	
    	if (status != DHS_S_SUCCESS)
    	{
		sprintf(errString,"saveDhsData: ERROR: unable to set instrument keyword .. expect the worst\n");
		printf(errString);
		goto DHSError;
		}
	
	
	/* always set the debug keyword */
	if (dx*dy> MAX_DMA)
		setHeaderDEBUG(bufNum,(unsigned int)reg.pXferStart,(unsigned int)(captBufAddr[bufNum]+ dy/2*dx),(unsigned int)(dhs->data));
	else
		setHeaderDEBUG(bufNum,(unsigned int)reg.pXferStart,0,(unsigned int)(dhs->data));
		
	if (dhs->headers != PARTIALHEAD) {
		/* build the DHS dataset header now */
		missing+=setDsHeader( ds,bufNum); 
		}
	
	if(status != DHS_S_SUCCESS)
	{
	    /* print error*/
	    printf("saveDhsData: unable to allocate new frame !!!\n");
	    goto DHSError;
	}
	
	/* create headers*/

	dhsBdAttribAdd (frame, "origin", DHS_DT_INT32, 1, ndims, origin, 
		    &status );

	
	if(status != DHS_S_SUCCESS)
	{
	    /* print error*/
	    printf("saveDhsData: unable to set origin attribute !!!\n");
	    goto DHSError;
	}

    dhsBdAttribAdd (frame, "axisSize", DHS_DT_INT32, 1, ndims, dims, 
		    &status );
	
	if(status != DHS_S_SUCCESS)
	{
	    /* print error*/
	    printf("saveDhsData: unable set axisSize attribute !!!\n");
	    goto DHSError;
	}

	/* write	 headers to dataset*/
	if(dhs_ok)
    	if (dhs->headers != PARTIALHEAD) {
			missing+=setFrameHeader( frame,bufNum);
			}

	if(status != DHS_S_SUCCESS)
	{
	    /* print error*/
	    printf("saveDhsData: frame new returned error\n");
	    goto DHSError;
	}

	/*set up coadder parameters*/
	gemLogMsg( 5 ,"dx = %d, dy = %d result = %x  run = %d\n",
					dx,dy,(unsigned)dhs->data,run);
#ifdef COADD_TEST
	vme = (char *)dhs->data;
#else
	sysLocalToBusAdrs(VME_AM_EXT_USR_DATA,(char *)(dhs->data),(char **)&vme);
#endif
	
/*
 *   printf("dma buffer = %x local = %x\n",
 *   	     (unsigned int)vme,(unsigned int)(dhs->data));
 */
	
		
	/* DMA mods #1 */
	
    reg.pDMAStart = (int)vme;
    reg.szDMABlock = dx * dy ;
    size = reg.numRows * reg.numCols;
    if (size > MAX_DMA) {	    
		reg.numRows = dy/2 ;
		gemLogMsg( 5 ,"dy = %d, numRows = %d\n",dy,reg.numRows);
		reg.szDMABlock = (reg.numRows) * (reg.numCols)  ;
    	}
    else
		reg.numRows = dy  ;

    semTake(coaddSem,WAIT_FOREVER);
    setRegisters(&reg,DMA);
    
	
/*
 * IS THIS MISSING TOO ???? only found late in sunSaver !!!
 *  header *phdr=NULL;
 *     phdr = dcHdrSet(headerVals[bufNum].ds,NULL);
 *     phdr = dcHdrSet(headerVals[bufNum].frame,phdr);
 *     if(phdr == NULL)
 *     {
 * 	printf("dcHdrset returned NULL\n");
 * 	goto FITSError;
 *     }
 */
	
	
	/* wait for completion of transfer*/
	
	if(semTake(sem.transFrame,WAIT_FOREVER) == ERROR) {
		printf("Error taking sem.transFrame semaphore\n");
		goto  DHSError;
		}

	if(size > MAX_DMA) {
        reg.numRows = dy - dy/2  ;
        reg.pDMAStart = (int)vme +(dy/2*dx*4);
        reg.numCols = dx ;
        reg.szDMABlock = (reg.numRows) * (reg.numCols)  ;
        gemLogMsg( 5 ,"dx = %d, dy = %d, size = %d\n",reg.numRows,
                reg.numCols,reg.szDMABlock);
        reg.pXferStart = captBufAddr[bufNum]+ dy/2*dx;

        setRegisters(&reg,DMA);	



        if(semTake(sem.transFrame,WAIT_FOREVER) == ERROR) {
            printf("saveDhsData:  sem take transFrame returned error\n");
            goto DHSError;
			}
		
		}

	if (dhs->headers != PARTIALHEAD) {
		
		}
	else {
		missing=0;
		}
	
 if(gnPutEpicsT(dbTop, DHSMISSING ".VAL", DCALONG, &missing)!= OK)
  		 printf ("failed to set DHSMISSING to %d\n",missing);
		
    semGive(coaddSem);
    
	/* thses is just a time flag, not error */
/*
 * 	if (dhs->headers != PARTIALHEAD)
 * 		rdBanCom635Time(-1,bufNum);
 */

    /* send data*/
    dx = proi->cols;
    dy = proi->rows;
    putEmptyBuf(bufNum);

	      printf("sending frame\n");

		if(p->pDMAStart != NULL)

		    ip = (int *)p->pDMAStart;

		    for (i = 0; i<p->szDMABlock;i++)

/******************************************************************************
 * setDsHeader
 ******************************************************************************/
int setDsHeader(DHS_BD_DATASET ds, int bufNum)
{
    int n;
    DHS_STATUS status;
    int all_status;
	
	all_status=0;
    status = DHS_S_SUCCESS;
	
    for (n=0;n<DHSDSHEADER;n++) {
		if (dsHdr[bufNum][n].status==OK) {
			if (dhsDsHeader[n].type==DCALONG)
				dhsBdAttribAdd( ds, dhsDsHeader[n].keyword, DHS_DT_INT32, 0, NULL, dsHdr[bufNum][n].value.lval,&status);
			else if(dhsDsHeader[n].type==DCADOUBLE)
				dhsBdAttribAdd( ds, dhsDsHeader[n].keyword, DHS_DT_DOUBLE, 0, NULL, dsHdr[bufNum][n].value.dval,&status);
			else if(dhsDsHeader[n].type==DCASTRING)
				dhsBdAttribAdd( ds, dhsDsHeader[n].keyword, DHS_DT_STRING, 0, NULL, dsHdr[bufNum][n].value.sval,&status);
			else
				printf ("header type not supported %d for %s\n",dhsDsHeader[n].type,dhsDsHeader[n].keyword);
			}
		else {
			gemLogMsg(6,"header %s [%d] has bad status [%d]\n",dhsDsHeader[n].keyword,n,dsHdr[bufNum][n].status);
			all_status++;
			}

	if(status != DHS_S_SUCCESS) {
		printf("ERROR: setDsHeader - unable to set header keyword %s\n",dhsDsHeader[n].keyword);
	    all_status++;
		}
	}  



    return all_status;
}
